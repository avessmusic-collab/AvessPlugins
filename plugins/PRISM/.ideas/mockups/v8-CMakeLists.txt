# ---- 3. Embed the UI (BinaryData::index_html) ----
# Stage 3 copies .ideas/mockups/v8-ui.html -> Source/ui/public/index.html.
# The page is fully self-contained (inline CSS/JS, no network fonts, no external
# scripts). It also requests /prism-init.js, which is generated at runtime by the
# resource provider (NOT embedded). The JUCE frontend library (js/juce/index.js)
# is not required: the page talks to window.__JUCE__.backend directly.
juce_add_binary_data(PRISM_UIResources
    SOURCES
        Source/ui/public/index.html
)

target_link_libraries(PRISM
    PRIVATE
        PRISM_UIResources
        juce::juce_gui_extra            # WebBrowserComponent
        juce::juce_dsp                  # juce::dsp::FFT (Stage 2)
        juce::juce_audio_utils
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags
)

# ---- 4. WebView compile definitions ----
target_compile_definitions(PRISM
    PUBLIC
        JUCE_WEB_BROWSER=1
        JUCE_USE_CURL=0
        JUCE_VST3_CAN_REPLACE_VST2=0
)

# ---- 5. Platform notes ----
if(WIN32)
    # WebView2 runtime backend. Static-link the loader so no WebView2Loader.dll has to ship.
    target_compile_definitions(PRISM PUBLIC JUCE_USE_WIN_WEBVIEW2_WITH_STATIC_LINKING=1)
elseif(APPLE)
    # WKWebView (WebKit) - nothing to configure.
elseif(UNIX)
    # Linux: WebKit2GTK must be installed (not a v1 target).
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WEBKIT2 REQUIRED webkit2gtk-4.1)
    target_link_libraries(PRISM PRIVATE ${WEBKIT2_LIBRARIES})
    target_include_directories(PRISM PRIVATE ${WEBKIT2_INCLUDE_DIRS})
endif()

# Rebuild the embedded UI when the HTML changes
set_property(TARGET PRISM_UIResources APPEND PROPERTY LINK_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/Source/ui/public/index.html")
